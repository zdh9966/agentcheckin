About the contract & risks

  • What it does: checkIn() can be called once per day. Check in the day after your last check-in and your streak grows by 1; miss a day and it restarts at 1. Every check-in emits an on-chain event, and the leaderboard is computed from those events.
  • The gate: Only addresses holding an Agent ID can check in. The contract checks balanceOf on the official registry 0x60b9262456de011f7d15a54cdd8c48baaa8b022f.
  • What it doesn't do: It takes no money, custodies no assets, has no token, and checking in promises no returns or airdrops. There is no admin backdoor: no owner, it can't be paused, it can't be upgraded — nobody can change the rules after deployment.
  • Cost: You pay the gas from your own wallet — about 0.00005–0.00014 BNB per check-in (a few cents).
  • Public record: Check-in records are fully public on-chain; anyone can see how many days an address has checked in.
  • Contract: 0x7E6D580a9E425a5b3eB3A395B31Ffff3B7896F36 (BSC Mainnet, deployed at block 134570129) · Deployment transaction

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

interface IAgentIdRegistry {
    function balanceOf(address account) external view returns (uint256);
}

contract AgentCheckIn {
    IAgentIdRegistry public immutable agentIdRegistry;

    mapping(address => uint256) public lastCheckInDay;
    mapping(address => uint256) public streak;
    mapping(address => uint256) public totalCheckIns;
    mapping(address => bool) public hasCheckedIn;
    uint256 public totalCheckers;

    event CheckIn(address indexed user, uint256 indexed day, uint256 streak);

    constructor(address _agentIdRegistry) {
        require(_agentIdRegistry != address(0), "registry is zero address");
        agentIdRegistry = IAgentIdRegistry(_agentIdRegistry);
    }

    function currentDay() public view returns (uint256) {
        return block.timestamp / 1 days;
    }

    function checkIn() external {
        require(
            agentIdRegistry.balanceOf(msg.sender) > 0,
            "AgentCheckIn: caller holds no Agent ID"
        );
        uint256 today = currentDay();
        require(
            lastCheckInDay[msg.sender] != today,
            "AgentCheckIn: already checked in today"
        );

        if (lastCheckInDay[msg.sender] == today - 1) {
            streak[msg.sender] += 1;
        } else {
            streak[msg.sender] = 1;
        }
        lastCheckInDay[msg.sender] = today;
        totalCheckIns[msg.sender] += 1;

        if (!hasCheckedIn[msg.sender]) {
            hasCheckedIn[msg.sender] = true;
            totalCheckers += 1;
        }

        emit CheckIn(msg.sender, today, streak[msg.sender]);
    }

    function getUser(address user)
        external
        view
        returns (uint256 _lastCheckInDay, uint256 _streak, uint256 _totalCheckIns)
    {
        return (lastCheckInDay[user], streak[user], totalCheckIns[user]);
    }
}

Troubleshooting
  • Tapping Connect does nothing: Most likely the page isn't open in the Binance App's built-in browser. Paste this page's link into the browser inside the Binance App and try again.
  • It says you have no Agent ID: This wallet never claimed an Agent ID (.ai name). Switch to a wallet that claimed one, or claim one in the Binance App first.
  • Already checked in today: That's normal — come back tomorrow. The day rolls over at 00:00 UTC.
  • The check-in transaction failed: If it never landed on-chain, it costs nothing. Read the failure reason, fix it, and tap check-in again.

This page collects and uploads nothing: connecting and checking in go straight from your wallet to the BSC chain, and the checker count and leaderboard are computed by reading public on-chain data.
